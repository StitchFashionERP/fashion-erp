import { NextResponse } from "next/server";
import { createClient } from "@/lib/supabase/server";

class ApiError extends Error {
  status: number;

  constructor(message: string, status = 400) {
    super(message);
    this.name = "ApiError";
    this.status = status;
  }
}

function asString(value: unknown) {
  return String(value ?? "").trim();
}

function asStringArray(value: unknown) {
  return Array.isArray(value)
    ? value.map(asString).filter(Boolean)
    : [];
}

function errorResponse(error: unknown) {
  if (error instanceof ApiError) {
    return NextResponse.json(
      { error: error.message },
      { status: error.status },
    );
  }

  return NextResponse.json(
    {
      error:
        error instanceof Error
          ? error.message
          : "De review kon niet worden opgeslagen.",
    },
    { status: 500 },
  );
}

async function getContext() {
  const supabase = await createClient();

  const {
    data: { user },
    error: userError,
  } = await supabase.auth.getUser();

  if (userError || !user) {
    throw new ApiError(
      "Je sessie is verlopen. Log opnieuw in.",
      401,
    );
  }

  const { data: memberships, error: membershipError } =
    await supabase
      .from("organization_members")
      .select("organization_id")
      .eq("user_id", user.id)
      .eq("active", true);

  if (membershipError) {
    throw new ApiError(membershipError.message, 500);
  }

  const organizationId = asString(
    memberships?.[0]?.organization_id,
  );

  if (!organizationId) {
    throw new ApiError(
      "Er is geen actieve organisatie gekoppeld.",
      403,
    );
  }

  return {
    supabase,
    userId: user.id,
    organizationId,
  };
}

export async function POST(
  request: Request,
  context: {
    params: Promise<{
      id: string;
    }>;
  },
) {
  try {
    const { id } = await context.params;
    const sourceJobId = asString(id);

    if (!sourceJobId) {
      throw new ApiError("Ongeldige AI Studio-job.");
    }

    const body = (await request
      .json()
      .catch(() => null)) as
      | {
          positivePoints?: unknown;
          improvementPoints?: unknown;
          comment?: unknown;
        }
      | null;

    const positivePoints = asStringArray(
      body?.positivePoints,
    );
    const improvementPoints = asStringArray(
      body?.improvementPoints,
    );
    const comment = asString(body?.comment);

    if (
      improvementPoints.length === 0 &&
      !comment
    ) {
      throw new ApiError(
        "Geef minimaal één verbeterpunt of opmerking op.",
      );
    }

    const {
      supabase,
      organizationId,
      userId,
    } = await getContext();

    const { data: sourceJob, error: sourceJobError } =
      await supabase
        .from("ai_studio_jobs")
        .select("*")
        .eq("organization_id", organizationId)
        .eq("id", sourceJobId)
        .maybeSingle();

    if (sourceJobError) {
      throw new ApiError(
        sourceJobError.message,
        500,
      );
    }

    if (!sourceJob) {
      throw new ApiError(
        "De oorspronkelijke AI-versie is niet gevonden.",
        404,
      );
    }

    if (
      sourceJob.status !== "COMPLETED" ||
      !sourceJob.result_path
    ) {
      throw new ApiError(
        "Alleen een voltooide packshot kan worden verbeterd.",
      );
    }

    const nextJobId = crypto.randomUUID();
    const now = new Date().toISOString();

    const feedbackPrompt = [
      "Create a new version based on the previous generated packshot.",
      "Preserve every product detail that was not explicitly mentioned.",
      positivePoints.length
        ? `Keep unchanged: ${positivePoints.join(", ")}.`
        : "",
      improvementPoints.length
        ? `Improve only these points: ${improvementPoints.join(", ")}.`
        : "",
      comment
        ? `Additional review comment: ${comment}`
        : "",
      "Do not redesign the garment.",
      "Do not change the composition, background or lighting unless specifically requested.",
    ]
      .filter(Boolean)
      .join("\n");

    const { data: nextJob, error: nextJobError } =
      await supabase
        .from("ai_studio_jobs")
        .insert({
          id: nextJobId,
          organization_id: organizationId,
          article_id: sourceJob.article_id,
          article_code: sourceJob.article_code,
          article_name: sourceJob.article_name,
          job_type: sourceJob.job_type,
          status: "CONCEPT",
          preset_name: sourceJob.preset_name,
          instructions: feedbackPrompt,
          source_bucket:
            sourceJob.result_bucket ??
            sourceJob.source_bucket,
          source_path:
            sourceJob.result_path ??
            sourceJob.source_path,
          source_file_name: `review-source-v${sourceJob.version_number ?? 1}.png`,
          source_mime_type: "image/png",
          source_file_size: 0,
          parent_job_id: sourceJobId,
          created_by: userId,
          created_at: now,
          updated_at: now,
        })
        .select("*")
        .single();

    if (nextJobError || !nextJob) {
      throw new ApiError(
        nextJobError?.message ??
          "De nieuwe AI-versie kon niet worden aangemaakt.",
        500,
      );
    }

    const { data: review, error: reviewError } =
      await supabase
        .from("ai_studio_reviews")
        .insert({
          organization_id: organizationId,
          source_job_id: sourceJobId,
          next_job_id: nextJobId,
          positive_points: positivePoints,
          improvement_points: improvementPoints,
          comment,
          status: "SUBMITTED",
          created_by: userId,
          created_at: now,
          updated_at: now,
        })
        .select("*")
        .single();

    if (reviewError || !review) {
      await supabase
        .from("ai_studio_jobs")
        .delete()
        .eq("organization_id", organizationId)
        .eq("id", nextJobId);

      throw new ApiError(
        reviewError?.message ??
          "De review kon niet worden opgeslagen.",
        500,
      );
    }

    return NextResponse.json(
      {
        reviewId: review.id,
        sourceJobId,
        nextJobId,
        nextVersionNumber:
          nextJob.version_number,
        feedbackPrompt,
      },
      { status: 201 },
    );
  } catch (error) {
    return errorResponse(error);
  }
}
